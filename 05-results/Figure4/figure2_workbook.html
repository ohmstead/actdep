I want to try my own heatmap implementation, instead of Seurat’s shitty one.

# get module assignment and then map module number to a color
module_assignments <- hclust_gene_mat |>
    scale() |>
    t() |>
    dist() |>
    hclust(method = 'complete') |>
    cutree(6)

df_gene_modules <- module_assignments |>
  as.data.frame() |> 
  rownames_to_column(var = 'gene') |>
  as_tibble() |> 
  rename(gene_module = module_assignments) |> 
  mutate(color = paletteer_d('ggthemes::wsj_colors6')[gene_module]) |> 
  mutate(gene_str = glue("<i style='color:{color}'>{gene}</i>"))

# make df for plotting
df_gg <- df_expression |> 
  filter(m_condition == 'd30m') |>
  filter(m_active_binary == FALSE) |>
  # arrange the row order of the plot before pivoting df
  arrange(m_supertype_name, -m_supertype_bootstrapping_probability) |> 
  pivot_longer(cols = -starts_with('m_'), names_to = 'gene', values_to = 'expression') |>
  left_join(df_gene_modules) # incorporate gene module info

```{r}
# get module assignment and then map module number to a color
module_assignments <- hclust_gene_mat |>
    scale() |>
    t() |>
    dist() |>
    hclust(method = 'complete') |>
    cutree(6)

df_gene_modules <- module_assignments |>
  as.data.frame() |> 
  rownames_to_column(var = 'gene') |>
  as_tibble() |> 
  rename(gene_module = module_assignments) |> 
  mutate(color = paletteer_d('ggthemes::wsj_colors6')[gene_module]) |> 
  mutate(gene_str = glue("<i style='color:{color}'>{gene}</i>"))

# make df for plotting
df_gg <- df_expression |> 
  filter(m_condition == 'd30m') |>
  filter(m_active_binary == FALSE) |>
  # arrange the row order of the plot before pivoting df
  arrange(m_supertype_name, -m_supertype_bootstrapping_probability) |> 
  pivot_longer(cols = -starts_with('m_'), names_to = 'gene', values_to = 'expression') |>
  left_join(df_gene_modules) # incorporate gene module info

order_by_module <- df_gg |>
  summarize(module_avg_expression = mean(expression), .by = c(m_cell, gene_module)) |>
  # make new variable which denotes which cluster for each cell has the highest average expression
  mutate(highest_module = which(module_avg_expression == max(module_avg_expression)), .by = m_cell) |> 
  arrange(highest_module, -module_avg_expression) |> 
  filter(gene_module == highest_module)

p_grid <- df_gg |> 
  left_join(order_by_module) |> 
ggplot() +
  aes(x = gene_str, y = m_cell, fill = expression) +
  geom_tile() +
  scale_y_discrete(limits = unique(order_by_module$m_cell)) +
  scale_fill_gradient2(low = '#075AFF', mid = 'white', high = '#FF0000', midpoint = 0, limits = c(-2.5, 2.5), oob = squish) +
